## Create a Next.js project

```sh
npx create-next-app@latest docs-site
cd docs-site
npm install @primer/doctocat-nextjs @primer/react-brand
```

Select TypeScript and App Router when prompted.

## Configure Next.js

```ts filename="next.config.ts"
import withDoctocat from '@primer/doctocat-nextjs/doctocat.config.js'

export default withDoctocat({
  transpilePackages: ['@primer/doctocat-nextjs'],
})
```

For a static export, also set `output: 'export'`, `trailingSlash: true`, and `images.unoptimized: true`.

## Add the root layout

```tsx filename="app/layout.tsx"
import Theme, {getPageMap} from '@primer/doctocat-nextjs'
import '@primer/doctocat-nextjs/css/global.css'
import type {ReactNode} from 'react'

export default async function RootLayout({children}: {children: ReactNode}) {
  const pageMap = await getPageMap()

  return (
    <html lang="en">
      <body>
        <Theme pageMap={pageMap}>{children}</Theme>
      </body>
    </html>
  )
}
```

## Add the content route

```tsx filename="app/[[...mdxPath]]/page.tsx"
import {generateStaticParamsFor, importPage} from '@primer/doctocat-nextjs'
import {notFound} from 'next/navigation'
import {useMDXComponents} from '../../mdx-components.js'

export const generateStaticParams = generateStaticParamsFor('mdxPath')
const {Article} = useMDXComponents()

export default async function Page({params}) {
  const {mdxPath} = await params

  try {
    const {default: Content, toc, metadata} = await importPage(mdxPath)
    return (
      <Article toc={toc} metadata={metadata}>
        <Content params={{mdxPath}} />
      </Article>
    )
  } catch {
    notFound()
  }
}
```

## Register MDX components

```js filename="mdx-components.js"
import {Article} from '@primer/doctocat-nextjs/components'

export function useMDXComponents(components = {}) {
  return {...components, Article}
}
```
